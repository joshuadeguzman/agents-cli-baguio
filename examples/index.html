Code review

Code reviewer

Reads the changes on your branch and reports bugs, security problems and missing tests, with file, line and severity.

Tools: get_diffread_file

Replace app/agent.py in a project made with agents-cli create code-reviewer --prototype --yes.

import subprocess
from pathlib import Path

from google.adk.agents import Agent
from google.adk.apps import App
from google.adk.models import Gemini
from google.genai import types

REPO = Path.cwd()


def get_diff(base: str = "main") -> dict:
    """Get the changes on this branch compared with a base branch.

    Call this first. It returns a unified diff with file names and line
    numbers, which is what you review.

    Args:
        base: The branch to compare against, for example "main".

    Returns:
        A dict with "status" and either "diff" or "error_message".
    """
    result = subprocess.run(
        ["git", "diff", f"{base}...HEAD", "--unified=3"],
        cwd=REPO, capture_output=True, text=True, timeout=30,
    )
    if result.returncode != 0:
        return {"status": "error", "error_message": result.stderr.strip()}
    return {"status": "success", "diff": result.stdout[:200_000]}


def read_file(path: str) -> dict:
    """Read a whole file from the repository, for context around a change.

    Args:
        path: A path relative to the repository root, for example "lib/main.dart".

    Returns:
        A dict with "status" and either "content" or "error_message".
    """
    file = (REPO / path).resolve()
    if REPO.resolve() not in file.parents or not file.is_file():
        return {"status": "error", "error_message": f"No file {path!r} in the repo."}
    return {"status": "success", "content": file.read_text(errors="replace")}


root_agent = Agent(
    name="code_reviewer",
    model=Gemini(model="gemini-3.8-flash", retry_options=types.HttpRetryOptions(attempts=3)),
    instruction=(
        "You review code changes. If the message contains a diff, review it. "
        "Otherwise call get_diff, and read_file when you need context. "
        "Report bugs, security problems and missing tests. For "
        "each finding give the file, the line, a severity (high, medium, low) "
        "and a one-line fix. If there is nothing worth fixing, say so."
    ),
    tools=[get_diff, read_file],
)

app = App(root_agent=root_agent, name="app")

Design review

Design reviewer

Looks at a screenshot of a mobile screen and lists the top five issues. Contrast and tap targets are measured with tools, not guessed.

Tools: contrast_ratiocheck_touch_target

Replace app/agent.py in a project made with agents-cli create design-reviewer --prototype --yes.

from google.adk.agents import Agent
from google.adk.apps import App
from google.adk.models import Gemini
from google.genai import types


def _luminance(hex_color: str) -> float:
    rgb = [int(hex_color.lstrip("#")[i:i + 2], 16) / 255 for i in (0, 2, 4)]
    lin = [c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4 for c in rgb]
    return 0.2126 * lin[0] + 0.7152 * lin[1] + 0.0722 * lin[2]


def contrast_ratio(foreground: str, background: str) -> dict:
    """Compute the WCAG contrast ratio between a text color and its background.

    Use this for every text color you can read from the screenshot. Do not
    guess contrast by eye.

    Args:
        foreground: Text color as hex, for example "#777777".
        background: Background color as hex, for example "#FFFFFF".

    Returns:
        A dict with the ratio and whether it passes WCAG AA for normal
        text (4.5:1) and large text (3:1).
    """
    a, b = _luminance(foreground), _luminance(background)
    ratio = (max(a, b) + 0.05) / (min(a, b) + 0.05)
    return {"ratio": round(ratio, 2), "aa_normal_text": ratio >= 4.5, "aa_large_text": ratio >= 3}


def check_touch_target(width_dp: float, height_dp: float) -> dict:
    """Check a tappable element against the 48 x 48 dp minimum touch target.

    Args:
        width_dp: Width of the element in dp.
        height_dp: Height of the element in dp.

    Returns:
        A dict with "passes" and the smallest side.
    """
    passes = width_dp >= 48 and height_dp >= 48
    return {"passes": passes, "smallest_side_dp": min(width_dp, height_dp)}


root_agent = Agent(
    name="design_reviewer",
    model=Gemini(model="gemini-3.8-flash", retry_options=types.HttpRetryOptions(attempts=3)),
    instruction=(
        "You review mobile app screens. The user attaches a screenshot. Check "
        "hierarchy, spacing and alignment by looking. Check text contrast with "
        "contrast_ratio and tap targets with check_touch_target, never by eye. "
        "Return the top five issues, most important first, each with a fix."
    ),
    tools=[contrast_ratio, check_touch_target],
)

app = App(root_agent=root_agent, name="app")

Release

Release notes writer

Turns the commits since your last tag into short release notes for app users, grouped into New, Improved and Fixed.

Tools: git_log

Replace app/agent.py in a project made with agents-cli create release-notes-writer --prototype --yes.

import subprocess

from google.adk.agents import Agent
from google.adk.apps import App
from google.adk.models import Gemini
from google.genai import types


def git_log(since: str) -> dict:
    """List the commits since a tag or commit, one per line.

    Call this first to see what changed in the release.

    Args:
        since: A tag or commit, for example "v1.4.0".

    Returns:
        A dict with "status" and either "commits" or "error_message".
    """
    result = subprocess.run(
        ["git", "log", f"{since}..HEAD", "--no-merges", "--pretty=format:%h %s"],
        capture_output=True, text=True, timeout=30,
    )
    if result.returncode != 0:
        return {"status": "error", "error_message": result.stderr.strip()}
    return {"status": "success", "commits": result.stdout.splitlines()}


root_agent = Agent(
    name="release_notes_writer",
    model=Gemini(model="gemini-3.8-flash", retry_options=types.HttpRetryOptions(attempts=3)),
    instruction=(
        "You write release notes for app users, not developers. If the message "
        "lists commits, use them. Otherwise call git_log with the tag the user "
        "gives. Group changes into New, Improved and "
        "Fixed. Skip refactors, CI and dependency bumps. Plain words, under "
        "500 characters."
    ),
    tools=[git_log],
)

app = App(root_agent=root_agent, name="app")
